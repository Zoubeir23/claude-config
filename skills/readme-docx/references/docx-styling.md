# .docx Styling Specifications

All Word documents generated by the readme-docx skill use these exact specs for visual
consistency across the entire docs/ folder.

Use `python-docx` to apply these programmatically. The generate_docs.py script should
define style helpers at the top and reuse them for every document.

---

## Fonts

| Element | Font | Size | Weight |
|---------|------|------|--------|
| Document title (title page) | Arial | 28pt | Bold |
| Heading 1 | Arial | 18pt | Bold |
| Heading 2 | Arial | 14pt | Bold |
| Heading 3 | Arial | 12pt | Bold |
| Normal text | Arial | 11pt | Regular |
| Code / monospace | Courier New | 10pt | Regular |
| Caption / note | Arial | 9pt | Italic |

## Document Structure

### Title Page

```python
# Title page layout (add page break after)
doc.add_paragraph()  # vertical spacer
title_para = doc.add_paragraph()
title_run = title_para.add_run(document_title)
title_run.font.name = "Arial"
title_run.font.size = Pt(28)
title_run.font.bold = True
title_run.font.color.rgb = RGBColor(0x2E, 0x75, 0xB6)
title_para.alignment = WD_ALIGN_PARAGRAPH.CENTER

subtitle_para = doc.add_paragraph(project_name)
subtitle_para.alignment = WD_ALIGN_PARAGRAPH.CENTER
# subtitle: Arial 14pt, gray color

date_para = doc.add_paragraph(f"Version {version} — {date}")
date_para.alignment = WD_ALIGN_PARAGRAPH.CENTER

doc.add_page_break()
```

### Code Blocks

```python
def add_code_block(doc, code_text, language=""):
    para = doc.add_paragraph()
    para.paragraph_format.space_before = Pt(12)
    para.paragraph_format.space_after = Pt(12)
    para.paragraph_format.left_indent = Inches(0.3)
    
    # Gray background
    pPr = para._p.get_or_add_pPr()
    shd = OxmlElement('w:shd')
    shd.set(qn('w:val'), 'clear')
    shd.set(qn('w:color'), 'auto')
    shd.set(qn('w:fill'), 'E8E8E8')
    pPr.append(shd)
    
    # Blue left border
    pBdr = OxmlElement('w:pBdr')
    left = OxmlElement('w:left')
    left.set(qn('w:val'), 'single')
    left.set(qn('w:sz'), '24')
    left.set(qn('w:space'), '4')
    left.set(qn('w:color'), '2E75B6')
    pBdr.append(left)
    pPr.append(pBdr)
    
    run = para.add_run(code_text)
    run.font.name = "Courier New"
    run.font.size = Pt(10)
```

### Tables

```python
def add_table(doc, headers, rows):
    table = doc.add_table(rows=1 + len(rows), cols=len(headers))
    table.style = 'Table Grid'
    
    # Header row — blue background
    header_cells = table.rows[0].cells
    for i, header in enumerate(headers):
        cell = header_cells[i]
        cell.text = header
        # Apply D5E8F0 background and bold font
        run = cell.paragraphs[0].runs[0]
        run.font.bold = True
        run.font.name = "Arial"
        run.font.size = Pt(10)
    
    # Data rows
    for row_idx, row_data in enumerate(rows):
        row_cells = table.rows[row_idx + 1].cells
        for i, value in enumerate(row_data):
            row_cells[i].text = str(value)
```

## Page Setup

```python
from docx.shared import Inches
from docx.oxml.ns import qn
from docx.oxml import OxmlElement

def setup_page(doc):
    section = doc.sections[0]
    section.page_width = Inches(8.5)
    section.page_height = Inches(11)
    section.left_margin = Inches(1)
    section.right_margin = Inches(1)
    section.top_margin = Inches(1)
    section.bottom_margin = Inches(1)
```
